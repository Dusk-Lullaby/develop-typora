# 09_并查集



## 9.1 概念

并查集（英文：Disjoint-set data structure， 直译为不交集数据结构）是一种数据结构，用于处理一些不交集的合并及查询问题。

* 查询(Find)：查询某个元素属于哪个集合，通常是返回集合内的一个“代表元素”。这个操作是为了判断两个元素是否在同一个集合之中。
* 合并(Union)：将两个集合合并为一个。
* 添加：添加一个新集合，其中有一个新元素。

## 9.2 `quickFind`

`QuickFind`方法：每个元素都有一个编号，只要编号一样，就认为是一个集团的人。合并时，把另外一个团伙的人的老大改为这个团伙的老大。

```c
//查找根
static int findRoot(QuickUnionSet* setQU, Element_t e) {
	int index = findIndex(setQU, e);
	if (index == -1) return -1;
	//当父节点是自己时，则为根节点
	while (index != setQU->parent[index]) {
		index = setQU->parent[index];
	}

	return index;
}

	int bRoot = findRoot(setQU, b);

	if (aRoot == -1 || bRoot == -1) return 0;
	return aRoot == bRoot;
}
```

### 9.3.7 路径压缩

#### 9.3.7.1 概念

路径压缩是并查集（Union-Find / Disjoint Set Union）算法中的一种关键优化策略。其主要目的是在执行 `find`（查找）操作的过程中，通过调整树的结构，减少树的高度，从而防止树形结构退化为链表，显著提升后续操作的效率。

在标准的 Quick Union 实现中，元素之间的关系被组织成树形结构。若缺乏优化，树的高度可能随操作次数线性增长（最坏情况 O(N)）。

**路径压缩的执行逻辑：**

当执行 `find(x)` 操作查找节点x的根节点时，算法会在递归回溯的过程中，将x到根节点路径上的**所有中间节点**的父指针（parent pointer），直接指向**根节点**。

这一过程将原本深层嵌套的树形结构“扁平化” (Flattening)，使得后续对路径上任意节点的查找操作均可在O(1)时间内完成。

可以用栈实现。

```c
//压栈
static SetNode* pushStack(SetNode* stack, int index) {
	SetNode* node = malloc(sizeof(SetNode));
	if (node == NULL) return NULL;
	node->index = index;
	node->parent = stack;
	stack = node;

	return stack;
}

//弹栈
static SetNode* popStack(SetNode* stack, int* index) {
	if (stack == NULL) return NULL;
	*index = stack->index;
	SetNode* node = stack;
	stack = stack->parent;
	free(node);
	return stack;
}

//查找根
static int findRoot(QuickUnionSet* setQU, Element_t e) {
	if (setQU == NULL) return -1;
	int curIndex = findIndex(setQU, e);
	SetNode* path = NULL;
	//遍历e号节点的父节点路径
	while (setQU->parent[curIndex] != curIndex) {
		path = pushStack(path, curIndex);
		curIndex = setQU->parent[curIndex];
	}
	//出栈
	while (path) {
		int pos;
		path = popStack(path, &pos);
		setQU->parent[pos] = setQU->parent[curIndex];
	}
	free(path);

	return curIndex;
}
```

```java
    //路径压缩
    //查找根
    private int findRoot(Element e) {
        return find(findIndex(e));
    }

    private int find(int curIndex) {
        if (this.parent[curIndex] == curIndex) {
            return curIndex;
        }
        // 递归逻辑：
        // 1. 先找到 parent[p] 的根节点
        // 2. 路径压缩：将 p 的父节点直接更新为这个根节点
        parent[curIndex] = find(parent[curIndex]);

        return parent[curIndex];
    }
```

```c
static SetNode* popStack(SetNode* stack, int* index) {
	if (stack == NULL) return NULL;
	SetNode* tmp = stack;
	stack = stack->parent;
	*index = tmp->index;
	free(tmp);
	return stack;
}

//路径压缩
//查找根
static int findRoot(QuickUnionSet* setQU, Element_t e) {
	if (setQU == NULL) return -1;
	int curIndex = findIndex(setQU, e);
	SetNode* path = NULL;
	//遍历e号节点的父节点路径
	while (setQU->parent[curIndex] != curIndex) {
		path = pushStack(path, setQU->data[curIndex]);
		curIndex = setQU->parent[curIndex];
	}
	//出栈，将这些节点的父节点不断指向根节点
	while (path) {
		int pos;
		path = popStack(path, &pos);
		setQU->parent[pos] = setQU->parent[curIndex];
	}
	free(path);

	return curIndex;
}
#endif

//查
/*
	1. 找到每个元素的索引，通过不断寻找父节点，找到其根编号
	2. 判断
*/
int isSameQU(QuickUnionSet* setQU, Element_t a, Element_t b) {
	if (setQU == NULL) return 0;
	int aRoot = findRoot(setQU, a);
	int bRoot = findRoot(setQU, b);
	if (aRoot == -1 || bRoot == -1) {
		return 0;
	}

	return aRoot == bRoot;
}

//并
void unionQU(QuickUnionSet* setQU, Element_t a, Element_t b) {
	if (setQU == NULL) return;
	int aRoot = findRoot(setQU, a);
	int bRoot = findRoot(setQU, b);
	if (aRoot == -1 || bRoot == -1 || aRoot == bRoot) 
		return;
	
	int aSize = setQU->size[aRoot];
	int bSize = setQU->size[bRoot];
	if (aSize > bSize) {
		setQU->parent[bRoot] = aRoot;
		setQU->size[aRoot] += bSize;
	}
	else {
		setQU->parent[aRoot] = bRoot;
		setQU->size[bRoot] += aSize;
	}
}
```

```java
            this.size[i] = 1;
            this.parent[i] = i;
        }
    }

    /*
        功能：查找
        参数：元素a 元素b
        返回值：成功true 失败false
     */

        int aRoot = findRoot(a);
        int bRoot = findRoot(b);
        if (aRoot == -1 || bRoot == -1) return false;
        return aRoot == bRoot;
    }

    /*
    //查找根
    private int findRoot(Element e) {
        int index = findIndex(e);
        if (index == -1) return index;
        //当父节点是自己时则为根节点
        while(this.parent[index] != index) {
            index = this.parent[index];
        }
        return index;
    }

     */

    //路径压缩
    //查找根
    private int findRoot(Element e) {
        return find(findIndex(e));
    }

    private int find(int curIndex) {
        if (this.parent[curIndex] == curIndex) {
            return curIndex;
        }
        // 递归逻辑：
        // 1. 先找到 parent[p] 的根节点
        // 2. 路径压缩：将 p 的父节点直接更新为这个根节点
        parent[curIndex] = find(parent[curIndex]);

        return parent[curIndex];
    }

    //查找索引
    private int findIndex(Element e) {
        for (int i = 0; i < this.data.length; i++) {
            if (this.data[i].equals(e))
                return i;
        }
        return -1;
    }

    /*
        功能：合并
        参数：元素a 元素b
        返回值：无
     */
    public void unionQuickUnionSet(Element a, Element b) {
        int aRoot = findRoot(a);
        int bRoot = findRoot(b);
        if (aRoot == -1 || bRoot == -1) return;
        if (this.size[aRoot] > this.size[bRoot]) {
            this.parent[bRoot] = this.parent[aRoot];
            this.size[aRoot] += this.size[bRoot];
        } else {
            this.parent[aRoot] = this.parent[bRoot];
            this.size[bRoot] += this.size[aRoot];
        }
    }
}
```

## 9.5 模板

```java
class UnionFind {
    private int[] parent;
    private int count; // 记录当前的连通块数量

    // 初始化并查集，n 为节点总数
    public UnionFind(int n) {
        // 通常竞赛中节点编号可能从 1 开始，所以数组大小开 n + 1 或 n + 5 防越界
        parent = new int[n + 5];
        count = n; 
        for (int i = 0; i <= n; i++) {
            parent[i] = i; // 初始状态：每个节点的父节点都是自己
        }
    }

    // 查找节点 x 的根节点（带路径压缩）
    // 时间复杂度：接近 O(1)
    public int find(int x) {
        if (parent[x] == x) {
            return x;
        }
        // 递归查找根节点，并将沿途节点直接指向根节点
        parent[x] = find(parent[x]);
        return parent[x];
    }

    // 合并 p 和 q 所在的两个集合
    public void union(int p, int q) {
        int rootP = find(p);
        int rootQ = find(q);

        // 如果根节点不同，说明属于不同集合，进行合并
        if (rootP != rootQ) {
            parent[rootP] = rootQ; // 将 p 的根节点挂载到 q 的根节点上
            count--;               // 连通块总数减 1
        }
    }

    // 判断 p 和 q 是否属于同一个集合（是否连通）
    public boolean isConnected(int p, int q) {
        return find(p) == find(q);
    }

    // 获取当前图中的连通块数量
    public int getCount() {
        return count;
    }
}
```

给定一个 n 个点、m 条边的无向图，求该图的连通块数量。 

输入：第一行点数 n 和边数 m。接下来 m 行，每行给出一条无向边 (p, q)。

<font color = "blue"> 数组解法：</font>

```java
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        // 开辟数组空间，+5 是为了防止索引越界（常用竞赛习惯）
        int[] parent = new int[n + 5];
        // 1. 初始化
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
        }

        // 2. 读入边并合并
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt();
            int q = sc.nextInt();
            union(parent, p, q);
        }

        // 3. 统计连通块数量（根节点数量）
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (find(parent, i) == i) {
                ans++;
            }
        }

        System.out.println(ans);
    }

    // 合并操作
    public static void union(int[] parent, int p, int q) {
        int roota = find(parent, p);
        int rootb = find(parent, q);

        // 建议补充此判断：避免在同一集合时执行自我赋值
        if (roota == rootb) return;

        // 将 a 所在的集合挂载到 b 所在的集合上
        parent[roota] = rootb; // 注意：由于上面赋值改为了 rootb，这里应该写 rootb 而不是 parent[rootb]
    }

    // 查找并进行路径压缩
    public static int find(int[] parent, int index) {
        // 找到根节点，直接返回
        if (parent[index] == index) return parent[index];

        // 递归寻找根节点，并执行路径压缩赋值
        parent[index] = find(parent, parent[index]);
        return parent[index];
    }
}
```

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    // 核心数据结构：Key 存当前节点，Value 存父节点
    static Map<Integer, Integer> parent = new HashMap<>();
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // 节点总数
        int m = sc.nextInt(); // 边数
        
        // 1. 初始化：将每个节点独立为一个连通块，父节点指向自己
        for (int i = 1; i <= n; i++) {
            parent.put(i, i);
        }
        
        // 2. 读入边并进行合并
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt();
            int q = sc.nextInt();
            union(p, q);
        }
        
        // 3. 统计连通块数量
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            // 只有根节点的特征是：find() 找出的根节点等于它自身
            if (i == find(i)) ans++;
        }
        
        System.out.println(ans);
    }
    
    // 合并操作
    public static void union(int p, int q) {
        int rootp = find(p); // 找 p 的根
        int rootq = find(q); // 找 q 的根
        
        // 如果根相同，说明已经在同一个连通块，直接跳过
        if (rootp == rootq) return;
        
        // 将 p 的根节点挂载到 q 的根节点上
        parent.put(rootp, rootq);
    }
    
    // 查找并进行路径压缩
    public static int find(int index) {
        int root = parent.get(index);
        
        // 递归终止条件：找到了根节点（父节点是自己）
        if (index == root) {
            return root;
        }
        
        // 递归寻找真正的根节点
        root = find(root);
        // 路径压缩：将当前节点直接指向真正的根节点，优化后续查询速度
        parent.put(index, root);
        return root;
    }
}
```

给定 n 个节点（编号 0 到 n-1）和无向边集 `edges`，判断节点 `source` 和 `destination` 之间是否连通。

```java
class Solution {
    int[] parent; // 使用数组替代 Map
    
    public boolean validPath(int n, int[][] edges, int source, int destination) {
        parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }
        
        for (int i = 0; i < edges.length; i++) {
            quickUnion(edges[i][0], edges[i][1]);
        }
        
        return findRoot(source) == findRoot(destination);
    }
    
    public void quickUnion(int a, int b) {
        int roota = findRoot(a);
        int rootb = findRoot(b);
        if (roota == rootb) return;
        parent[rootb] = roota; // 直接数组赋值
    }
    
    public int findRoot(int key) {
        if (parent[key] == key) return key;
        
        parent[key] = findRoot(parent[key]); // 路径压缩
        return parent[key];
    }
}
```

> 时间复杂度：`O(m + n)`
