# 十字链表

对于有向图，邻接表和逆邻接表并非全能，为了方便找出度，又方便找入度，可以把邻接表和逆邻接表合二为一。

十字链表只能存储有向图



## 1. 申明

<font color = "blue"> C 实现：</font>

```c
// 弧结构
typedef struct arcBox {
	int head;		// 弧头
	int tail;		// 弧尾
	struct arcBox* headNext;	// 下一个弧头
	struct arcBox* tailNext;	// 下一个弧尾
	int weight;		// 权重
}ArcBox;

// 顶点结构
typedef struct {
	int number;
	char* show;
	ArcBox* firstIn;	// 第一个入度
	ArcBox* firstOut;	// 第一个出度
}ArcVertex;

// 十字链表结构
typedef struct {
	ArcVertex* nodes;
	int vertexNum;
	int edegNum;
}CrossGraph;

```

```java
public class CrossLinkGraph<T> {

    // 弧结构
    private static class ArcBox {
        int head;               // 弧头
        int tail;               // 弧尾
        int weight;             // 权重
        ArcBox headNext;        // 下一个弧头
        ArcBox tailNext;        // 下一个弧尾
    }
    
    // 顶点结构
    private static class ArcVertex<T> {
        int number;         // 位置
        ArcBox firstIn;     // 入度
        ArcBox firstOut;    // 出度

    }
    
    private int vertxNum;           // 顶点数量
    private int edgeNum;            // 弧的数量
    private ArcVertex[] vertices;   // 顶点数组

}

```

```c
void addArcBox(CrossGraph* graph, int head, int tail, int w) {
	ArcBox* box = malloc(sizeof(ArcBox));

	box->head = head;
	box->tail = tail;
```

## 7. 释放

<font color = "blue"> C 实现：</font>

```c
// 释放
void releaseCrossGraph(CrossGraph* graph) {
	if (graph == NULL) return;
	for (int i = 0; i < graph->vertexNum; i++) {
		ArcBox* p = graph->nodes[i].firstOut;
		while (p) {
			ArcBox* tmp = p->tailNext;
			free(p);
			p = tmp;
		}
	}

	free(graph->nodes);
	free(graph);
}
```

## 8. 完整实现

<font color = "blue"> C 实现：</font>

`crossLinkGraph.h`

```c
#pragma once

// 弧结构
typedef struct arcBox {
	int tailVertex;				// 弧尾编号，出度
	struct arcBox* tailNext;	// 下一个弧尾
	int headVertex;				// 弧头编号，入度
	struct arcBox* headNext;	// 下一个弧头
	int weight;					// 权重
}ArcBox;

// 十字链表的顶点结构
typedef struct {
	int number;
	const char* show;
	ArcBox* firstIn;	// 入度
	ArcBox* firstOut;	// 出度
}CrossVertex;

// 图头 顺序存储实现图的结构
typedef struct {
	CrossVertex* nodes;
	int numVertex;
	int numEdge;
}CrossGraph;

/*
	功能：产生n个节点的十字链表的图
	参数：顶点的个数
	返回值：图
*/
CrossGraph* createCrossGraph(int n);

/*
	功能：释放图
	参数：图
	返回值：无
*/
void releaseCrossGraph(CrossGraph* graph);

/*
	功能：初始化图，设置顶点信息
	参数：图
	返回值：无
*/
void initCrossGraph(CrossGraph* graph, int num, char* names[]);

/*
	功能：添加边
	参数：图 弧头 弧尾 权重
	返回值：无
*/
void addCrossArc(CrossGraph* graph, int head, int tail, int w);

/*
	功能：计算number顶点入度
	参数：图 第几个顶点
	返回值：入度
*/
int inDegreeCrossGraph(CrossGraph* graph, int number);

/*
	功能：计算number顶点出度
	参数：图 第几个顶点
	返回值：出度
*/
int outDegreeCrossGraph(CrossGraph* graph, int number);
```

`crossLinkGraph.c`

```c
#include <stdio.h>
#include <stdlib.h>
#include "crossLinkGraph.h"

// 创建
CrossGraph* createCrossGraph(int n) {
	CrossGraph* graph = malloc(sizeof(CrossGraph));
	if (graph == NULL) return NULL;
	graph->numVertex = n;
	graph->numEdge = 0;
	graph->nodes = malloc(sizeof(CrossVertex) * n);
	if (graph->nodes == NULL) {
		free(graph);
		return NULL;
	}
	return graph;
}


// 初始化
void initCrossGraph(CrossGraph* graph, int num, char* names[]) {
	if (graph == NULL) return;
	for (int i = 0; i < num; i++) {
		graph->nodes[i].show = names[i];
		graph->nodes[i].number = i;
		graph->nodes[i].firstOut = NULL;
		graph->nodes[i].firstIn = NULL;
	}
}

// 添加
void addCrossArc(CrossGraph* graph, int head, int tail, int w) {
	ArcBox* box = malloc(sizeof(ArcBox));
	if (box == NULL) return;
	box->headVertex = head;
	box->tailVertex = tail;
	box->weight = w;

	// 头插法 (尾 -> 头)
	// 入度关系
	box->headNext = graph->nodes[head].firstIn;
	graph->nodes[head].firstIn = box;
	// 出度关系
	box->tailNext = graph->nodes[tail].firstOut;
	graph->nodes[tail].firstOut = box;

	graph->numEdge++;
}

// 计算节点入度
int inDegreeCrossGraph(CrossGraph* graph, int numbre) {
	int count = 0;
	ArcBox* box = graph->nodes[numbre].firstIn;
	while (box) {
		box = box->headNext;
		count++;
	}

	return count;
}

// 计算节点出度
int outDegreeCrossGraph(CrossGraph* graph, int number) {
	int count = 0;
	ArcBox* box = graph->nodes[number].firstOut;
	while (box) {
		box = box->headNext;
		count++;
	}

	return count;
}

// 释放
void releaseCrossGraph(CrossGraph* graph) {
	if (graph == NULL) return;

	for (int i = 0; i < graph->numVertex; i++) {
		ArcBox* p = graph->nodes[i].firstOut;
		while (p) {
			ArcBox* tmp = p->tailNext;
			free(p);
			p = tmp;
		}
	}
	free(graph->nodes);
	free(graph);
}

```

`main.c`

```c
#include <stdio.h>
#include <stdlib.h>
#include "crossLinkGraph.h"

void setupGraph(CrossGraph* graph) {
	char* nodeNames[] = { "V0", "V1", "V2", "V3" };
	initCrossGraph(graph, sizeof(nodeNames) / sizeof(nodeNames[0]), nodeNames);

	// (尾 -> 头)
	addCrossArc(graph, 3, 0, 1);
	addCrossArc(graph, 0, 1, 1);
	addCrossArc(graph, 0, 2, 1);
	addCrossArc(graph, 2, 1, 1);
	addCrossArc(graph, 1, 2, 1);
}

int main() {
	int n = 4;
	CrossGraph* graph = createCrossGraph(n);
	if (graph == NULL) return -1;
	setupGraph(graph);

	printf("V0 inDegree: %d\n", inDegreeCrossGraph(graph, 0));
	printf("V0 outDegree: %d\n", outDegreeCrossGraph(graph, 0));
	printf("graph edge number : %d\n", graph->numEdge);

	releaseCrossGraph(graph);

	return 0;
}
```

```java
package com.Sonnt.cross_link_graph;

import org.w3c.dom.html.HTMLParagraphElement;

public class CrossLinkGraph<T> {

    // 弧结构
    private static class ArcBox {
        int head;               // 弧头
        int tail;               // 弧尾
        int weight;             // 权重
        ArcBox headNext;        // 下一个弧头
        ArcBox tailNext;        // 下一个弧尾

        public ArcBox(int head, int tail, int weight) {
            this.head = head;
            this.tail = tail;
            this.weight = weight;
        }
    }

    // 顶点结构
    private static class ArcVertex<T> {
        int number;         // 位置
        ArcBox firstIn;     // 入度
        ArcBox firstOut;    // 出度
        T show;             // 顶点数据

        public ArcVertex(int number, T show) {
            this.number = number;
            this.firstIn = null;
            this.firstOut = null;
            this.show = show;
        }
    }

    private int vertxNum;           // 顶点数量
    private int edgeNum;            // 弧的数量
    private ArcVertex[] vertices;   // 顶点数组

    public CrossLinkGraph(int vertxNum) {
        this.vertxNum = vertxNum;
        this.edgeNum = 0;
        this.vertices = new ArcVertex[vertxNum];
    }

    /*
        功能：初始化
        参数：顶点数组数据
        返回值：无
     */
    public void initCrossGraph(T[] names) {
        for (int i = 0; i < this.vertxNum; i++) {
            this.vertices[i] = new ArcVertex(i, names[i]);
        }
    }

    /*
        功能：添加边
        参数：弧头 弧尾 权重
        返回值：无
     */
    public void addArcBox(int head, int tail, int weight) {
        ArcBox box = new ArcBox(head, tail, weight);
        box.headNext = this.vertices[head].firstIn;
        this.vertices[head].firstIn = box;
        box.tailNext = this.vertices[tail].firstOut;
        this.vertices[tail].firstOut = box;

        this.edgeNum++;
    }

    /*
        功能：计算出度
        参数：顶点位置
        返回值：出度
     */
    public int outDegreeVertex(int num) {
        int count = 0;
        ArcBox box = this.vertices[num].firstOut;
        while (box != null) {
            box = box.tailNext;
            count++;
        }

        return count;
    }

    /*
        功能：计算入度
        参数：顶点位置
        返回值：出度
     */
    public int inDegreeVertex(int num) {
        int count = 0;
        ArcBox box = this.vertices[num].firstIn;
        while (box != null) {
            box = box.headNext;
            count++;
        }

        return count;
    }

    /*
        功能：释放
        参数：无
        返回值：无
     */
    public void releaseCrossGraph() {
        for (int i = 0; i < this.vertxNum; i++) {
            this.vertices[i] = null;
        }

        this.vertices = null;
        this.edgeNum = 0;
        this.vertxNum = 0;
    }

    public int getEdgeNum() {
        return edgeNum;
    }
}
```
